
Pointer :
	pointer are variable that contain address.pointer start with * symbol.
	variable value assign to pointer with & symbol.
	pointer variable are declared as :
			Syn:
			    data type *variable_name;
	variable assign to pointer:
		ptr_name=&var_name;

		Ex1:
		      ptr=&num;
		Ex2:
		     int a,*ptr=&a;


	#include<stdio.h>

	{
	  int x,*ptr;
	  ptr=&x;
	  printf(" Enter an integer :");
	  scanf("%d",ptr);
	  printf("\n The value : %d",x);
	  getch();
	}

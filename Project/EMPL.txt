void payslip()
{
  f1=fopen("per.emp","rb+");
  f2=fopen("pay.emp","wb+");
  while(fread(&e,sizeof(e),1,f1)==1)
  {
    strcpy(p.ename,e.name);
    p.eid=e.id;
    p.bpay=e.bp;
    p.da=0.1*p.bpay;
    p.hra=0.2*p.bpay;
    p.ded=0.4*p.bpay;
    p.gp=p.bpay+p.da+p.hra;
    p.np=p.gp-p.ded;
    fwrite(&p,sizeof(p),1,f2);
    tempbp=0;
  }
  fclose(f1);
  fclose(f2);
}

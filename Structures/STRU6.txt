#include<stdio.h>
#include<conio.h>
struct s1
{
	char *str;
	struct s1 *next;
};
void main()
{
	static struct s1 arr[]={
			       {"Anbu",arr+1},{"Nandhu",arr+2},{"Anand",arr+3}};
	struct s1 *p[3];
	int i;
	clrscr();
	for(i=0;i<=2;i++)
		p[i]=arr[i].next;
	printf("%s %s %s",p[0]->str,*p->str,**p.str);
	swap(*p,arr);
	printf("\n%s",p[0]->str);
	printf("\n%s",(*p)->str);
	printf("\n%s",(*p)->next->str);
	swap(p[0],p[0]->str);
	printf("\n%s",p[0]->str);
	printf("\n%s",(*++p[0]).str);
	printf("\n%s",++(*++(*p)->next).str);
	getch();
}
